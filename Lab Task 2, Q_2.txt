# Question No. 2: Advanced Student Marksheet

name = input("Enter student's name: ")
roll_no = input("Enter roll number: ")

marks = []

# Input marks for 5 subjects
for i in range(1, 6):
    mark = float(input(f"Enter marks for Subject {i}: "))
    marks.append(mark)

# Calculate total and percentage
total = sum(marks)
percentage = total / 5

# Determine grade
if percentage >= 80:
    grade = "A+"
elif percentage >= 70:
    grade = "A"
elif percentage >= 60:
    grade = "B"
elif percentage >= 50:
    grade = "C"
elif percentage >= 40:
    grade = "D"
else:
    grade = "F"

# Check pass/fail
if any(mark < 40 for mark in marks):
    result = "Fail"
else:
    result = "Pass"

# Display marksheet
print("\n========== STUDENT MARKSHEET ==========")
print("Name       :", name)
print("Roll No.   :", roll_no)

for i in range(5):
    print(f"Subject {i + 1} :", marks[i])

print("----------------------------------------")
print("Total Marks:", total, "/ 500")
print("Percentage :", percentage, "%")
print("Grade      :", grade)
print("Result     :", result)
print("========================================")
